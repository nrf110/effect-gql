This example demonstrates modular GraphQL schema composition, service-based
architecture, and Effect DataLoaders using in-memory data.
The files below form a runnable compact version. Append the `services.ts`
continuations in order. The repository example also includes additional queries
and mutations, such as publishing posts and creating comments.

<Aside type="caution">
The demo authenticates every request as Alice, an administrator. It is not
production authentication. Before deployment, replace `AuthServiceLive` with
request-scoped authentication that validates credentials, and replace the
in-memory stores with persistent storage.
</Aside>

## Running the Example

```bash
pnpm example:full
# Server starts at http://localhost:4003
# GraphiQL available at http://localhost:4003/graphiql
```

## Domain Models

Define your domain with Effect Schema. These serve as the single source of truth:

```typescript
// domain.ts
import { Effect } from "effect"
import * as S from "effect/Schema"

export const UserRole = S.Literals(["ADMIN", "USER", "GUEST"])
export type UserRole = S.Schema.Type<typeof UserRole>

export const User = S.Struct({
  id: S.String,
  name: S.String,
  email: S.String,
  role: UserRole,
})
export type User = S.Schema.Type<typeof User>

export const CreateUserInput = S.Struct({
  name: S.String.check(S.isMinLength(1), S.isMaxLength(100)),
  email: S.String.check(S.isPattern(/^[^\s@]+@[^\s@]+\.[^\s@]+$/)),
  role: UserRole.pipe(S.withDecodingDefaultKey(Effect.succeed("USER" as const))),
})
export type CreateUserInput = S.Schema.Type<typeof CreateUserInput>

export const Post = S.Struct({
  id: S.String,
  title: S.String,
  content: S.String,
  authorId: S.String,
  published: S.Boolean,
  createdAt: S.Number,
})
export type Post = S.Schema.Type<typeof Post>

export const CreatePostInput = S.Struct({
  title: S.String.check(S.isMinLength(1)),
  content: S.String,
  published: S.optional(S.Boolean),
})
export type CreatePostInput = S.Schema.Type<typeof CreatePostInput>

export const Comment = S.Struct({
  id: S.String,
  content: S.String,
  postId: S.String,
  authorId: S.String,
  createdAt: S.Number,
})
export type Comment = S.Schema.Type<typeof Comment>
```

<Aside type="tip">
Effect Schema provides built-in validation. The `CreateUserInput` schema validates email format and name length automatically.
</Aside>

## Services

Services encapsulate business logic and external dependencies:

```typescript
// services.ts
import { Effect, Context, Layer, Ref } from "effect"
import type { User, Post, Comment, CreateUserInput, CreatePostInput } from "./domain"
import { NotFoundError, AuthorizationError } from "@effect-gql/core"

export class AuthService extends Context.Service<
  AuthService,
  {
    readonly getCurrentUser: () => Effect.Effect<User | null>
    readonly requireAuth: () => Effect.Effect<User, AuthorizationError>
    readonly requireRole: (role: string) => Effect.Effect<User, AuthorizationError>
  }
>()("AuthService") {}

export class UserService extends Context.Service<
  UserService,
  {
    readonly findById: (id: string) => Effect.Effect<User, NotFoundError>
    readonly findByIds: (ids: readonly string[]) => Effect.Effect<readonly User[]>
    readonly findAll: () => Effect.Effect<readonly User[]>
    readonly create: (input: CreateUserInput) => Effect.Effect<User>
  }
>()("UserService") {}

export class PostService extends Context.Service<
  PostService,
  {
    readonly findById: (id: string) => Effect.Effect<Post, NotFoundError>
    readonly findByAuthorIds: (ids: readonly string[]) => Effect.Effect<readonly Post[]>
    readonly findAll: () => Effect.Effect<readonly Post[]>
    readonly create: (authorId: string, input: CreatePostInput) => Effect.Effect<Post>
  }
>()("PostService") {}
```

```typescript
// services.ts (continued)
export class CommentService extends Context.Service<CommentService, {
  readonly findByPostIds: (ids: readonly string[]) => Effect.Effect<readonly Comment[]>
}>()("CommentService") {}
```

### Service Implementations

```typescript
// services.ts (continued)
const usersRef = Ref.makeUnsafe<readonly User[]>([
  { id: "1", name: "Alice", email: "alice@example.com", role: "ADMIN" },
  { id: "2", name: "Bob", email: "bob@example.com", role: "USER" },
])
const postsRef = Ref.makeUnsafe<readonly Post[]>([
  {
    id: "p1", title: "Getting Started with Effect", content: "Welcome!",
    authorId: "1", published: true, createdAt: Date.now(),
  },
])
const commentsRef = Ref.makeUnsafe<readonly Comment[]>([
  { id: "c1", content: "Great article!", postId: "p1", authorId: "2", createdAt: Date.now() },
])
let userId = 2
let postId = 1

const getCurrentUser = () => Ref.get(usersRef).pipe(
  Effect.map((users) => users[0] ?? null)
)
const requireAuth = () => Effect.gen(function* () {
  const user = yield* getCurrentUser()
  if (!user) {
    return yield* Effect.fail(new AuthorizationError({ message: "Authentication required" }))
  }
  return user
})

export const AuthServiceLive = Layer.succeed(AuthService, {
  getCurrentUser,
  requireAuth,
  requireRole: (role) => Effect.gen(function* () {
    const user = yield* requireAuth()
    if (user.role !== role) {
      return yield* Effect.fail(new AuthorizationError({ message: `${role} role required` }))
    }
    return user
  }),
})

export const UserServiceLive = Layer.succeed(UserService, {
  findById: (id) => Effect.gen(function* () {
    const users = yield* Ref.get(usersRef)
    const user = users.find((candidate) => candidate.id === id)
    if (!user) {
      return yield* Effect.fail(
        new NotFoundError({ message: "User not found", resource: `User:${id}` })
      )
    }
    return user
  }),
  findByIds: (ids) => Ref.get(usersRef).pipe(
    Effect.map((users) => users.filter((user) => ids.includes(user.id)))
  ),
  findAll: () => Ref.get(usersRef),
  create: (input) => Effect.gen(function* () {
    const user: User = { ...input, id: String(++userId) }
    yield* Ref.update(usersRef, (users) => [...users, user])
    return user
  }),
})

export const PostServiceLive = Layer.succeed(PostService, {
  findById: (id) => Effect.gen(function* () {
    const posts = yield* Ref.get(postsRef)
    const post = posts.find((candidate) => candidate.id === id)
    if (!post) {
      return yield* Effect.fail(
        new NotFoundError({ message: "Post not found", resource: `Post:${id}` })
      )
    }
    return post
  }),
  findByAuthorIds: (ids) => Ref.get(postsRef).pipe(
    Effect.map((posts) => posts.filter((post) => ids.includes(post.authorId)))
  ),
  findAll: () => Ref.get(postsRef).pipe(
    Effect.map((posts) => posts.filter((post) => post.published))
  ),
  create: (authorId, input) => Effect.gen(function* () {
    const post: Post = {
      ...input, id: `p${++postId}`, authorId,
      published: input.published ?? false, createdAt: Date.now(),
    }
    yield* Ref.update(postsRef, (posts) => [...posts, post])
    return post
  }),
})

export const CommentServiceLive = Layer.succeed(CommentService, {
  findByPostIds: (ids) => Ref.get(commentsRef).pipe(
    Effect.map((comments) => comments.filter((comment) => ids.includes(comment.postId)))
  ),
})

export const ServicesLive = Layer.mergeAll(
  AuthServiceLive, UserServiceLive, PostServiceLive, CommentServiceLive
)
```

## DataLoaders

Define loaders that integrate with your services:

```typescript
// loaders.ts
import { Effect } from "effect"
import { Loader } from "@effect-gql/core"
import { UserService, PostService, CommentService } from "./services"
import type { User, Post, Comment } from "./domain"

export const loaders = Loader.define({
  UserById: Loader.single<string, User, UserService>({
    batch: (ids) =>
      Effect.gen(function* () {
        const userService = yield* UserService
        return yield* userService.findByIds(ids)
      }),
    key: (user) => user.id,
  }),

  PostsByAuthorId: Loader.grouped<string, Post, PostService>({
    batch: (authorIds) =>
      Effect.gen(function* () {
        const postService = yield* PostService
        return yield* postService.findByAuthorIds(authorIds)
      }),
    groupBy: (post) => post.authorId,
  }),
  CommentsByPostId: Loader.grouped<string, Comment, CommentService>({
    batch: (ids) => Effect.gen(function* () {
      const service = yield* CommentService
      return yield* service.findByPostIds(ids)
    }),
    groupBy: (comment) => comment.postId,
  }),
})
```

## Schema Composition

Compose the schema using the pipe API:

```typescript
// schema.ts
import { Effect } from "effect"
import * as S from "effect/Schema"
import {
  GraphQLSchemaBuilder,
  query,
  mutation,
  objectType,
  enumType,
  field,
} from "@effect-gql/core"
import { UserRole, User, Post, Comment, CreateUserInput, CreatePostInput } from "./domain"
import { AuthService, UserService, PostService } from "./services"
import { loaders } from "./loaders"

export const schema = GraphQLSchemaBuilder.empty
  .pipe(
    // Register types
    enumType({ name: "UserRole", values: ["ADMIN", "USER", "GUEST"] }),
    objectType({ name: "User", schema: User }),
    objectType({ name: "Post", schema: Post }),
    objectType({ name: "Comment", schema: Comment }),
  )
  .pipe(

    // Computed fields with loaders
    field("User", "posts", {
      type: S.Array(Post),
      resolve: (parent: User) => loaders.load("PostsByAuthorId", parent.id),
    }),

    field("Post", "author", {
      type: User,
      resolve: (parent: Post) => loaders.load("UserById", parent.authorId),
    }),

    field("Post", "comments", {
      type: S.Array(Comment),
      resolve: (parent: Post) => loaders.load("CommentsByPostId", parent.id),
    }),
    field("Post", "commentCount", {
      type: S.Int,
      resolve: (parent: Post) => loaders.load("CommentsByPostId", parent.id).pipe(
        Effect.map((comments) => comments.length)
      ),
    }),
    field("Comment", "author", {
      type: User,
      resolve: (parent: Comment) => loaders.load("UserById", parent.authorId),
    }),
  )
  .pipe(

    // Queries
    query("me", {
      type: S.NullOr(User),
      resolve: () =>
        Effect.gen(function* () {
          const auth = yield* AuthService
          return yield* auth.getCurrentUser()
        }),
    }),

    query("users", {
      type: S.Array(User),
      resolve: () =>
        Effect.gen(function* () {
          const userService = yield* UserService
          return [...(yield* userService.findAll())]
        }),
    }),

    query("posts", {
      type: S.Array(Post),
      resolve: () => Effect.gen(function* () {
        const service = yield* PostService
        return [...(yield* service.findAll())]
      }),
    }),

    // Mutations with auth
    mutation("createUser", {
      args: CreateUserInput,
      type: User,
      resolve: (args) =>
        Effect.gen(function* () {
          const auth = yield* AuthService
          yield* auth.requireRole("ADMIN")

          const userService = yield* UserService
          return yield* userService.create(args)
        }),
    }),
    mutation("createPost", {
      args: CreatePostInput,
      type: Post,
      resolve: (input) => Effect.gen(function* () {
        const auth = yield* AuthService
        const user = yield* auth.requireAuth()
        const service = yield* PostService
        return yield* service.create(user.id, input)
      }),
    }),
  )
  .buildSchema()
```

## Server Entry Point

Bring everything together:

```typescript
// index.ts
import { Layer } from "effect"
import { HttpRouter, HttpServerResponse } from "effect/http"
import { makeGraphQLRouter } from "@effect-gql/core"
import { serve } from "@effect-gql/node"

import { schema } from "./schema"
import { ServicesLive } from "./services"
import { loaders } from "./loaders"

// Supply loader dependencies, keeping application services available too.
const AppLayer = loaders.toLayer().pipe(Layer.provideMerge(ServicesLive))

// Create router
const graphqlRouter = makeGraphQLRouter(schema, AppLayer, {
  path: "/graphql",
  graphiql: { path: "/graphiql", endpoint: "/graphql" },
})

const app = Layer.mergeAll(
  HttpRouter.add("GET", "/health", HttpServerResponse.json({ status: "ok" })),
  graphqlRouter
)

// Start server
serve(app, Layer.empty, { port: 4003 })
```

      }
    }
  }

### Create a Post

The example simulates a logged-in user (Alice with ADMIN role), so mutations work without explicit authentication headers.

```graphql
mutation {
  createPost(title: "My New Post", content: "Hello world!") {
    id
    title
    author {
      name
    }
  }
}
```
